#!/bin/bash
# 1. Compress your Unity WebGL build output directory
tar -czf build.tar.gz -C ./BuildOutputFolder .

# 2. Upload the compressed bundle to your VM
gcloud compute scp build.tar.gz instance-20250801-113609:/tmp/build.tar.gz \
    --zone=us-central1-c \
    --project=perfect-altar-409523

# 3. Connect via SSH to unpack the build directly into your web root
gcloud compute ssh instance-20250801-113609 \
    --zone=us-central1-c \
    --project=perfect-altar-409523 \
    --command="sudo tar -xzf /tmp/build.tar.gz -C /var/www/html && rm /tmp/build.tar.gz"

echo "Deployment complete!"
